# SQL 性能分析

## MySQL

```mysql
SELECT * FROM
(SELECT
QUERY_SAMPLE_TEXT,
DIGEST_TEXT,
COUNT_STAR ,
ROUND( MAX_TIMER_WAIT/1000/1000/1000/1000 ) MAX_TIMER_WAIT_S,
ROUND( AVG_TIMER_WAIT/1000/1000/1000/1000 ) AVG_TIMER_WAIT_S,
ROUND(SUM_LOCK_TIME/COUNT_STAR/1000/1000/1000/1000) AVG_LOCK_TIME_S,
ROUND(SUM_ROWS_EXAMINED/COUNT_STAR) AVG_ROWS_EXAMINED,
ROUND(SUM_ROWS_SENT/COUNT_STAR) AVG_ROWS_SENT,
ROUND(SUM_ROWS_AFFECTED/COUNT_STAR) AVG_ROWS_AFFECTED,
ROUND((SUM_ROWS_SENT + SUM_ROWS_AFFECTED)/ COUNT_STAR) AVG_ROWS_SENT_AFFECTED,
FIRST_SEEN,
LAST_SEEN
FROM
PERFORMANCE_SCHEMA.EVENTS_STATEMENTS_SUMMARY_BY_DIGEST
WHERE
COUNT_STAR>10 AND (SCHEMA_NAME IS NULL OR SCHEMA_NAME NOT IN ('SYS', 'MYSQL', 'INFORMATION_SCHEMA', 'PERFORMANCE_SCHEMA'))
AND QUERY_SAMPLE_TEXT NOT LIKE '%APPLICATIONNAME=DBEAVER%'
AND QUERY_SAMPLE_TEXT NOT LIKE '%LIMIT 0,200%'
AND QUERY_SAMPLE_TEXT NOT LIKE '%ATT_APIT EXECUTE%'
-- AND QUERY_SAMPLE_TEXT NOT LIKE '/*+TSN=%'
AND QUERY_SAMPLE_TEXT NOT LIKE '%INFORMATION_SCHEMA%'
AND (AVG_TIMER_WAIT/1000/1000/1000/1000>3)
) T
-- WHERE DIGEST_TEXT LIKE '%IEAI_DEVOPS_PARAM_INSTANCE%'
ORDER BY COUNT_STAR DESC
LIMIT 100;

SELECT * FROM PERFORMANCE_SCHEMA.EVENTS_STATEMENTS_SUMMARY_BY_DIGEST WHERE QUERY_SAMPLE_TEXT LIKE '/*';
```
